// Game card component \u2014 visual64 treatment: each adventure fronted by its real 64-bit place art
interface GameCardProps {
  title: string
  description: string
  href: string
  icon: string
  art?: string
  available: boolean
  comingSoon?: boolean
  isNew?: boolean
  locked?: boolean
  lockHint?: string
  features?: string[]
}

function GameCard({ title, description, href, icon, art, available, comingSoon, isNew, locked, lockHint, features }: GameCardProps) {
  const isAccessible = available && !locked
  const content = (
    <div
      className={`
        relative overflow-hidden rounded-lg border transition-all duration-300 group
        ${isAccessible
          ? 'border-[#9a7a48]/60 bg-[#171b1b]/95 shadow-[0_12px_32px_rgba(0,0,0,0.4)] hover:border-amber-400 hover:shadow-[0_16px_40px_rgba(212,169,80,0.15)] cursor-pointer'
          : locked
            ? 'border-purple-700/60 bg-[#14121a]/95 cursor-not-allowed opacity-80'
            : 'border-stone-700 bg-[#141616]/95 cursor-not-allowed opacity-60'}
      `}
    >
      {art && (
        <div className="relative aspect-video">
          <Image
            src={art}
            alt={title}
            fill
            className={`object-cover transition-transform duration-500 ${isAccessible ? 'group-hover:scale-105' : 'grayscale-[0.6] brightness-75'}`}
            sizes="(max-width: 640px) 50vw, 260px"
          />
          <div className="absolute inset-0 bg-[linear-gradient(180deg,transparent_45%,rgba(19,20,20,0.95))]" aria-hidden="true" />
          <span className="absolute bottom-1.5 left-2 text-xl drop-shadow-[0_2px_4px_rgba(0,0,0,0.9)]">{locked ? '\uD83D\uDD12' : icon}</span>
        </div>
      )}
      {comingSoon && (
        <div className="absolute top-2 right-2 z-10 bg-red-700/90 text-white text-[8px] font-pixel px-2 py-1 rounded">
          Coming Soon
        </div>
      )}
      {isNew && !comingSoon && !locked && (
        <div className="absolute top-2 right-2 z-10 bg-emerald-700/90 text-white text-[8px] font-pixel px-2 py-1 rounded animate-pulse">
          NEW!
        </div>
      )}
      {locked && (
        <div className="absolute top-2 right-2 z-10 bg-purple-800/90 text-purple-100 text-[8px] font-pixel px-2 py-1 rounded">
          Locked
        </div>
      )}
      <div className="p-3">
        {!art && (
          <div className="text-3xl mb-2 text-center group-hover:scale-110 transition-transform">
            {locked ? '\uD83D\uDD12' : icon}
          </div>
        )}
        <h3 className="font-pixel text-[#f0d69a] text-[11px] leading-4 mb-1.5">{title}</h3>
        <p className="text-stone-400 text-[10px] leading-4">{description}</p>
        {locked && lockHint && (
          <p className="text-purple-300/70 text-[9px] mt-2 italic">{lockHint}</p>
        )}
        {features && features.length > 0 && (
          <div className="mt-2 flex flex-wrap gap-1">
            {features.map((feature, i) => (
              <span key={i} className="text-[8px] border border-[#9a7a48]/40 bg-black/40 text-amber-200/90 px-1.5 py-0.5 rounded">
                {feature}
              </span>
            ))}
          </div>
        )}
      </div>
    </div>
  )

  if (isAccessible) {
    return <Link href={href}>{content}</Link>
  }
  return content
}
